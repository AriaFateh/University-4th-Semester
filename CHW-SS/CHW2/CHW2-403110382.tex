	 $$a(1) y(n) = \sum_{m = 1}^{N_b} b(m) x(n-m+1) - \sum_{m = 2}^{N_a} a(m) y(n-m+1)$$}

	\subsection{Theory}
	\begin{itemize}
		\item 
		$y_1 = m(t) \cos(2\pi f_1t)^2$ :
		$$\cos(\omega_1 t)^2 = \frac12(1 + \cos(2 \omega_1 t))$$
		$$\mathcal{F}(y_1(t)) = \frac12 M(j\omega) + \frac14 \left(M(j(\omega - 2\omega_1)) +M(j(\omega + 2\omega_1))\right)$$
		\item 
		$y_2(t) = m(t) \cos(\omega_1) \sin(\omega_1 t)$ :
		$$y_2(t) = \frac12 m(t) \sin(2\omega_1 t)$$
		$$\mathcal{F}\{y_2(t)\} = \frac{1}{4j} \left(M(j(\omega - 2\omega_1)) - M(j(\omega + 2\omega_1))\right)$$
		\item 
		$y_3(t) = m(t) \cos(\omega_1 t) \cos(\omega_2 t)$ :
		$$y_3(t) = \frac12 m(t)\cos((\omega_1-\omega_2)t)+\frac12 m(t)\cos((\omega_1+\omega_2)t)$$
		\begin{align*}
		\mathcal{F}\{y_3(t)\} &=\frac14\Big[
		M(j(\omega-(\omega_1-\omega_2)))+M(j(\omega+(\omega_1-\omega_2)))\\
		+&M(j(\omega-(\omega_1+\omega_2)))+M(j(\omega+(\omega_1+\omega_2)))
		\Big]	
		\end{align*}
	\end{itemize}
